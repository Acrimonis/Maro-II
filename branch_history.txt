/**
 * Full-screen overlay displaying a LazyColumn of track summary cards.
 *
 * Each card shows:
 * - Date, name (editable inline), comment (editable inline)
 * - Max speed, distance, point count
 * - Visibility toggle, GPX share, delete (with confirmation dialog)
 *
 * @param trackSummaries List of track summaries to display.
 * @param onUpdateTrack Called to update track metadata (name, comment, visibleOnMap).
 * @param onDeleteTrack Called to delete a track by ID.
 * @param onShareGpx Called with the track ID to share GPX.
 * @param onDismiss Called to close the overlay.
 */
@Composable
fun TrackHistoryOverlay(
    trackSummaries: List<TrackSummary>,
    onUpdateTrack: (String, name: String?, comment: String?, visibleOnMap: Boolean?) -> Unit,
    onDeleteTrack: (String) -> Unit,
    onShareGpx: (String) -> Unit,
    onDismiss: () -> Unit,
    modifier: Modifier = Modifier
) {
    var deleteConfirmId by remember { mutableStateOf<String?>(null) }

    Box(
        modifier = modifier
            .fillMaxSize()
            .background(Color(0xFF1A1A2E))
            .windowInsetsPadding(WindowInsets.statusBars)
    ) {
        Column(modifier = Modifier.fillMaxSize()) {
            // ── Header ────────────────────────────────────────────────────
            Row(
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(16.dp),
                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.CenterVertically
            ) {
                Button(
                    onClick = onDismiss,
                    shape = CircleShape,
                    colors = ButtonDefaults.buttonColors(
                        containerColor = Color(0x33FFFFFF)
                    ),
                    modifier = Modifier.size(48.dp),
                    contentPadding = androidx.compose.foundation.layout.PaddingValues(0.dp)
                ) {
                    Text(
                        text = "\u2190",
                        color = Color.White,
                        fontSize = 22.sp
                    )
                }
                Text(
                    text = "Track History",
                    color = Color.White,
                    fontSize = 22.sp,
                    fontWeight = FontWeight.Bold
                )
                Spacer(modifier = Modifier.width(48.dp)) // spacer for symmetry
            }

            // ── Track list ────────────────────────────────────────────────
            LazyColumn(
                modifier = Modifier
                    .fillMaxSize()
                    .padding(horizontal = 16.dp),
                verticalArrangement = Arrangement.spacedBy(12.dp)
            ) {
                items(trackSummaries, key = { it.id }) { summary ->
                    TrackCard(
                        summary = summary,
                        onUpdateName = { name -> onUpdateTrack(summary.id, name, null, null) },
                        onUpdateComment = { comment -> onUpdateTrack(summary.id, null, comment, null) },
                        onToggleVisibility = {
                            onUpdateTrack(summary.id, null, null, !summary.visibleOnMap)
                        },
                        onShare = { onShareGpx(summary.id) },
                        onDelete = { deleteConfirmId = summary.id },
                        isDeleteConfirming = deleteConfirmId == summary.id,
                        onConfirmDelete = { onDeleteTrack(summary.id); deleteConfirmId = null },
                        onCancelDelete = { deleteConfirmId = null }
                    )
                }
            }
        }
    }
}

@Composable
private fun TrackCard(
    summary: TrackSummary,
    onUpdateName: (String) -> Unit,
    onUpdateComment: (String) -> Unit,
    onToggleVisibility: () -> Unit,
    onShare: () -> Unit,
    onDelete: () -> Unit,
    isDeleteConfirming: Boolean,
    onConfirmDelete: () -> Unit,
    onCancelDelete: () -> Unit
) {
    val dateFormat = remember { SimpleDateFormat("yyyy-MM-dd HH:mm", Locale.getDefault()) }
    val startDate = remember(summary.startTimeMs) { dateFormat.format(Date(summary.startTimeMs)) }

    Column(
        modifier = Modifier
            .fillMaxWidth()
            .clip(RoundedCornerShape(12.dp))
            .background(Color(0x1AFFFFFF))
            .padding(12.dp)
    ) {
        // ── Date and visibility toggle ────────────────────────────────────
        Row(
            modifier = Modifier.fillMaxWidth(),
            horizontalArrangement = Arrangement.SpaceBetween,
            verticalAlignment = Alignment.CenterVertically
        ) {
            Text(
                text = startDate,
                color = Color(0xFF1565C0),
                fontSize = 14.sp,
                fontWeight = FontWeight.Bold
            )
            Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                // Visibility toggle
                Button(
                    onClick = onToggleVisibility,
                    shape = CircleShape,
                    colors = ButtonDefaults.buttonColors(
                        containerColor = if (summary.visibleOnMap)
                            Color(0xFF1565C0).copy(alpha = 0.3f)
                        else Color(0x33FFFFFF)
                    ),
                    modifier = Modifier.size(36.dp),
                    contentPadding = androidx.compose.foundation.layout.PaddingValues(0.dp)
                ) {
                    Text(
                        text = if (summary.visibleOnMap) "\uD83D\uDC41" else "\uD83D\uDC41\u200D\uD83D\uDDE8",
                        fontSize = 16.sp
                    )
                }
            }
        }

        Spacer(modifier = Modifier.height(8.dp))

        // ── Editable name ─────────────────────────────────────────────────
        var editingName by remember(summary.id) { mutableStateOf(false) }
        var nameText by remember(summary.id, summary.name) { mutableStateOf(summary.name) }

        if (editingName) {
            TextField(
                value = nameText,
                onValueChange = { nameText = it },
                singleLine = true,
                colors = TextFieldDefaults.colors(
                    focusedTextColor = Color.White,
                    unfocusedTextColor = Color.White,
                    focusedContainerColor = Color(0x33FFFFFF),
                    unfocusedContainerColor = Color(0x1AFFFFFF)
                ),
                modifier = Modifier.fillMaxWidth()
            )
            Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                Button(
                    onClick = { onUpdateName(nameText); editingName = false },
                    colors = ButtonDefaults.buttonColors(containerColor = Color(0xFF2E7D32))
                ) { Text("Save", color = Color.White, fontSize = 12.sp) }
                Button(
                    onClick = { editingName = false; nameText = summary.name },
                    colors = ButtonDefaults.buttonColors(containerColor = Color(0x33FFFFFF))
                ) { Text("Cancel", color = Color.White, fontSize = 12.sp) }
            }
        } else {
            Text(
                text = summary.name,
                color = Color.White,
                fontSize = 16.sp,
                fontWeight = FontWeight.Medium,
                modifier = Modifier
                    .fillMaxWidth()
                    .clickable(
                        interactionSource = remember { MutableInteractionSource() },
                        indication = null
                    ) { editingName = true }
            )
        }

        // ── Editable comment ──────────────────────────────────────────────
        var editingComment by remember(summary.id) { mutableStateOf(false) }
        var commentText by remember(summary.id, summary.comment) { mutableStateOf(summary.comment) }

        if (editingComment) {
            TextField(
                value = commentText,
                onValueChange = { commentText = it },
                singleLine = true,
                keyboardOptions = KeyboardOptions(imeAction = ImeAction.Done),
                keyboardActions = KeyboardActions(
                    onDone = { onUpdateComment(commentText); editingComment = false }
                ),
                colors = TextFieldDefaults.colors(
                    focusedTextColor = Color.White,
                    unfocusedTextColor = Color.White,
                    focusedContainerColor = Color(0x33FFFFFF),
                    unfocusedContainerColor = Color(0x1AFFFFFF)
                ),
                modifier = Modifier.fillMaxWidth()
            )
            Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                Button(
                    onClick = { onUpdateComment(commentText); editingComment = false },
                    colors = ButtonDefaults.buttonColors(containerColor = Color(0xFF2E7D32))
                ) { Text("Save", color = Color.White, fontSize = 12.sp) }
                Button(
                    onClick = { editingComment = false; commentText = summary.comment },
                    colors = ButtonDefaults.buttonColors(containerColor = Color(0x33FFFFFF))
                ) { Text("Cancel", color = Color.White, fontSize = 12.sp) }
            }
        } else {
            Text(
                text = if (summary.comment.isNotEmpty()) summary.comment else "Add comment...",
                color = if (summary.comment.isNotEmpty()) Color.White else Color(0x66FFFFFF),
                fontSize = 14.sp,
                modifier = Modifier
                    .fillMaxWidth()
                    .clickable(
                        interactionSource = remember { MutableInteractionSource() },
                        indication = null
                    ) { editingComment = true }
            )
        }

        Spacer(modifier = Modifier.height(8.dp))

        // ── Stats ─────────────────────────────────────────────────────────
        Row(
            modifier = Modifier.fillMaxWidth(),
            horizontalArrangement = Arrangement.spacedBy(16.dp)
        ) {
            val speedKn = summary.fastestSpeedMps * 1.94384f
            StatChip("Max", "${"%.1f".format(speedKn)} kn")
            StatChip("Dist", "%.2f nm".format(summary.distanceNm))
            summary.endTimeMs?.let { endMs ->
                val durationSec = (endMs - summary.startTimeMs) / 1000
                StatChip("Dur", formatDurationShort(durationSec))
            }
        }

        Spacer(modifier = Modifier.height(8.dp))

        // ── Action buttons ────────────────────────────────────────────────
        if (isDeleteConfirming) {
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.spacedBy(8.dp)
            ) {
                Button(
                    onClick = onConfirmDelete,
                    colors = ButtonDefaults.buttonColors(containerColor = Color(0xFFC62828)),
                    modifier = Modifier.weight(1f)
                ) {
                    Text("Confirm Delete", color = Color.White, fontSize = 13.sp)
                }
                Button(
                    onClick = onCancelDelete,
                    colors = ButtonDefaults.buttonColors(containerColor = Color(0x33FFFFFF)),
                    modifier = Modifier.weight(1f)
                ) {
                    Text("Cancel", color = Color.White, fontSize = 13.sp)
                }
            }
        } else {
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.spacedBy(8.dp)
            ) {
                Button(
                    onClick = onShare,
                    colors = ButtonDefaults.buttonColors(containerColor = Color(0xFF1565C0)),
                    modifier = Modifier.weight(1f)
                ) {
                    Text("\uD83D\uDCE4 GPX", color = Color.White, fontSize = 13.sp)
                }
                Button(
                    onClick = onDelete,
                    colors = ButtonDefaults.buttonColors(containerColor = Color(0xFFC62828).copy(alpha = 0.4f)),
                    modifier = Modifier.weight(1f)
                ) {
                    Text("\uD83D\uDDD1\uFE0F Delete", color = Color(0xFFEF9A9A), fontSize = 13.sp)
                }
            }
        }
    }
}

@Composable
private fun StatChip(label: String, value: String) {
    Column(horizontalAlignment = Alignment.CenterHorizontally) {
        Text(
            text = value,
            color = Color.White,
            fontSize = 14.sp,
            fontWeight = FontWeight.Bold
        )
        Text(
            text = label,
            color = Color(0xFF90A4AE),
            fontSize = 11.sp
        )
    }
}

private fun formatDurationShort(totalSeconds: Long): String {
    val hours = totalSeconds / 3600
    val minutes = (totalSeconds % 3600) / 60
    val seconds = totalSeconds % 60
    return when {
        hours > 0 -> "${hours}h${minutes}m"
        minutes > 0 -> "${minutes}m${seconds}s"
        else -> "${seconds}s"
    }
}
